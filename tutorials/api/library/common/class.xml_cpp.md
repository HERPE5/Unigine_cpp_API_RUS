# Unigine::Xml Class (CPP)

**Header:** #include <UnigineXml.h>


Поскольку Unigine широко использует XML для хранения различных типов данных, движок также предоставляет функциональность для разбора и сохранения файлов этого типа.


### Пример


Пример ниже создаёт Xml и выводит все добавленные данные в консоль.


```cpp
#include "AppWorldLogic.h"

using namespace Unigine;

XmlPtr my_xml_create()
{
	// создание xml с корневым элементом
	XmlPtr xml = Xml::create("xml_element");

	// добавление дочерних элементов
	XmlPtr xml_0 = xml->addChild("child", "arg=\"0\"");
	XmlPtr xml_1 = xml_0->addChild("child1", "arg=\"1\"");
	XmlPtr xml_2 = xml_1->addChild("child2", "arg=\"2\"");

	// установка данных для дочернего элемента
	xml_2->setData("data");

	return xml;
}

// метод для вывода данных Xml в консоль
void my_xml_print(XmlPtr xml, int offset=0)
{
	// создание отступов
	for (int i = 0; i < offset; i++)
		Log::message(" ");

	// вывод информации xml
	Log::message("%s: ", xml->getName());

	for (int i = 0; i < xml->getNumArgs(); i++)
		Log::message("%s=%s ", xml->getArgName(i), xml->getArgValue(i));

	Log::message(": %s\n", xml->getData());

	for (int i = 0; i < xml->getNumChildren(); i++)
		my_xml_print(XmlPtr(xml->getChild(i)), offset + 1);
}

int AppWorldLogic::init()
{

	// создание дерева xml
	XmlPtr xml = my_xml_create();

	// вывод дерева xml
	my_xml_print(xml);

	return 1;
}

```
